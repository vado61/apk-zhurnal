/** * АПК — приём данных из калькулятора инспектора в Google Таблицу. * Развернуть как веб-приложение: Deploy -> New deployment -> Web app * Execute as: Me | Who has access: Anyone * * Логика ID: каждая запись (машина / смена) создаётся в калькуляторе с уникальным ID. * Если время убытия/окончания смены известно не сразу — строка уходит в таблицу * неполной, а когда инспектор дозаполнит её в калькуляторе, скрипт находит строку * по ID (колонка A, скрыта) и дописывает нужные ячейки — вместо создания новой строки. */ var JOURNAL_HEADERS = [ 'ID записи','Метка времени отправки','Дата','Объект','№ по порядку','Время прибытия', 'Марка машины','Гос. номер','Водитель (ФИО)','№ ТТН','Дата ТТН', 'Характер груза','Количество груза','Ед. измерения','Получатель груза', 'Время убытия','Инспектор (ФИО)','Продолжительность на объекте' ]; var SHIFT_HEADERS = [ 'ID записи','Инспектор (ФИО)','Объект','Дата','Начало смены','Окончание смены','Длительность смены' ]; // Запустить один раз вручную из редактора (выбрать функцию setup -> Выполнить) function setup() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var journal = ss.getSheetByName('Журнал') || ss.insertSheet('Журнал'); journal.clear(); journal.appendRow(JOURNAL_HEADERS); journal.setFrozenRows(1); journal.getRange(1,1,1,JOURNAL_HEADERS.length).setFontWeight('bold'); journal.hideColumns(1); // скрываем колонку ID — инспекторам она не нужна var shifts = ss.getSheetByName('Смены') || ss.insertSheet('Смены'); shifts.clear(); shifts.appendRow(SHIFT_HEADERS); shifts.setFrozenRows(1); shifts.getRange(1,1,1,SHIFT_HEADERS.length).setFontWeight('bold'); shifts.hideColumns(1); if (!ss.getSheetByName('Сводка')) ss.insertSheet('Сводка'); // Защита листа "Журнал" — редактировать сможет только владелец (то есть скрипт) var protection = journal.protect().setDescription('Журнал — только скрипт'); protection.removeEditors(protection.getEditors()); if (protection.canDomainEdit()) protection.setDomainEdit(false); SpreadsheetApp.getUi().alert('Готово: листы созданы, колонка ID скрыта, "Журнал" защищён.'); } function doPost(e) { var result = { status: 'ok' }; try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); switch (data.type) { case 'entry_new': { var jSheet = ss.getSheetByName('Журнал'); var dur = calcDuration(data.arriveTime, data.departTime); jSheet.appendRow([ data.id, new Date(), toDateObj(data.logDate), data.object || '', data.order || '', data.arriveTime || '', data.carBrand || '', data.carPlate || '', data.driverName || '', data.ttnNumber || '', toDateObj(data.ttnDate), data.cargoType || '', data.cargoQty || '', data.cargoUnit || '', data.cargoReceiver || '', data.departTime || '', data.inspector || '', dur ]); jSheet.getRange(jSheet.getLastRow(), 2).setNumberFormat('dd.mm.yy hh:mm'); jSheet.getRange(jSheet.getLastRow(), 3).setNumberFormat('dd.mm.yy'); jSheet.getRange(jSheet.getLastRow(), 11).setNumberFormat('dd.mm.yy'); break; } case 'entry_update_departure': { var jSheet2 = ss.getSheetByName('Журнал'); var rowIdx = findRowById(jSheet2, data.id); if (rowIdx > 0) { var arriveVal = jSheet2.getRange(rowIdx, 6).getValue(); // F = Время прибытия var dur2 = calcDuration(formatSheetTime(arriveVal), data.departTime); jSheet2.getRange(rowIdx, 16).setValue(data.departTime || ''); // Время убытия jSheet2.getRange(rowIdx, 18).setValue(dur2); // Продолжительность } else { result = { status: 'error', message: 'Запись с таким ID не найдена в Журнале' }; } break; } case 'shift_start': { var shSheet = ss.getSheetByName('Смены'); shSheet.appendRow([ data.id, data.inspector || '', data.object || '', toDateObj(data.date), data.start || '', '', '' ]); shSheet.getRange(shSheet.getLastRow(), 4).setNumberFormat('dd.mm.yy'); break; } case 'shift_end': { var shSheet2 = ss.getSheetByName('Смены'); var rowIdx2 = findRowById(shSheet2, data.id); if (rowIdx2 > 0) { var startVal = shSheet2.getRange(rowIdx2, 5).getValue(); // E = Начало смены var dur3 = calcDuration(formatSheetTime(startVal), data.end); shSheet2.getRange(rowIdx2, 6).setValue(data.end || ''); shSheet2.getRange(rowIdx2, 7).setValue(dur3); } else { result = { status: 'error', message: 'Смена с таким ID не найдена' }; } break; } default: result = { status: 'error', message: 'Неизвестный тип запроса: ' + data.type }; } } catch (err) { result = { status: 'error', message: String(err) }; } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } // Ищет строку по ID в скрытой колонке A. Возвращает номер строки (1-based) или -1. function findRowById(sheet, id) { if (!id) return -1; var lastRow = sheet.getLastRow(); if (lastRow < 2) return -1; var ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues(); for (var i = 0; i < ids.length; i++) { if (String(ids[i][0]) === String(id)) return i + 2; } return -1; } function calcDuration(t1, t2) { if (!t1 || !t2) return ''; var a = String(t1).split(':'), b = String(t2).split(':'); var minutes = (parseInt(b[0])*60 + parseInt(b[1])) - (parseInt(a[0])*60 + parseInt(a[1])); if (isNaN(minutes) || minutes < 0) return ''; var h = Math.floor(minutes/60), m = minutes % 60; return (h < 10 ? '0'+h : h) + ':' + (m < 10 ? '0'+m : m); } // Google Таблицы иногда хранят время как объект Date — приводим к "HH:MM" function formatSheetTime(val) { if (!val) return ''; if (val instanceof Date) { return Utilities.formatDate(val, Session.getScriptTimeZone(), 'HH:mm'); } return String(val); } // Превращает "2026-08-22" в настоящую дату (а не текст), чтобы работало форматирование ДД.ММ.ГГ function toDateObj(dateStr) { if (!dateStr) return ''; var d = new Date(dateStr + 'T00:00:00'); return isNaN(d.getTime()) ? '' : d; } // Разовая функция: чинит уже внесённые тестовые строки (текстовые даты -> настоящие даты + формат ДД.ММ.ГГ) // Запустить один раз вручную: выбрать "migrateAndFormatDates" -> Выполнить function migrateAndFormatDates() { var ss = SpreadsheetApp.getActiveSpreadsheet(); function fixColumn(sheet, col) { var lastRow = sheet.getLastRow(); if (lastRow < 2) { sheet.getRange(2, col, 5000, 1).setNumberFormat('dd.mm.yy'); return; } var range = sheet.getRange(2, col, lastRow - 1, 1); var values = range.getValues(); for (var i = 0; i < values.length; i++) { var v = values[i][0]; if (typeof v === 'string' && /^\d{4}-\d{2}-\d{2}$/.test(v)) { values[i][0] = new Date(v + 'T00:00:00'); } } range.setValues(values); // форматируем и уже заполненные, и запас строк на будущее sheet.getRange(2, col, 5000, 1).setNumberFormat('dd.mm.yy'); } var journal = ss.getSheetByName('Журнал'); fixColumn(journal, 3); // C — Дата fixColumn(journal, 11); // K — Дата ТТН journal.getRange(2, 2, 5000, 1).setNumberFormat('dd.mm.yy hh:mm'); // B — Метка времени отправки var shifts = ss.getSheetByName('Смены'); fixColumn(shifts, 4); // D — Дата SpreadsheetApp.getUi().alert('Готово: даты переведены в формат ДД.ММ.ГГ (в т.ч. на будущие строки).'); } // Проверка, что веб-приложение отвечает (открой URL в браузере) + отдача записей калькулятору function doGet(e) { var action = e.parameter.action; if (action === 'fetchEntries') { return fetchEntries(e.parameter.object, e.parameter.date); } return ContentService.createTextOutput('Скрипт работает. Отправка данных — только через POST из калькулятора.'); } // Отдаёт калькулятору уже внесённые записи по объекту и дате — чтобы на новом // устройстве/у другого инспектора локальная таблица не выглядела пустой. function fetchEntries(object, date) { var result = { status: 'ok', entries: [] }; try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('Журнал'); var lastRow = sheet.getLastRow(); if (lastRow >= 2) { var data = sheet.getRange(2, 1, lastRow - 1, 18).getValues(); for (var i = 0; i < data.length; i++) { var row = data[i]; var rowDateStr = formatSheetDate(row[2]); // C — Дата var rowObject = row[3]; // D — Объект if (rowObject === object && rowDateStr === date) { result.entries.push({ id: row[0], logDate: rowDateStr, object: row[3], order: row[4], arriveTime: row[5], carBrand: row[6], carPlate: row[7], driverName: row[8], ttnNumber: row[9], ttnDate: formatSheetDate(row[10]), // K — Дата ТТН cargoType: row[11], cargoQty: row[12], cargoUnit: row[13], cargoReceiver: row[14], departTime: row[15], inspector: row[16] }); } } } } catch (err) { result = { status: 'error', message: String(err) }; } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } // Приводит дату ячейки (может быть Date-объектом) к формату "yyyy-MM-dd" для сравнения/передачи function formatSheetDate(val) { if (!val) return ''; if (val instanceof Date) { return Utilities.formatDate(val, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } return String(val); }
